## Decyzje projektowe

1. **Baseline ma trzy wpisy `board`, nie dwa**: `services/boardService → boardTypes`, `services/boardService → boardSchema`, `app/api/board/route → boardSchema`. Wszystkie znikają razem z kodem; baseline czyści osobny commit (zasada z CLAUDE.md i spec §6).
2. **Walidacja wejścia w `router`, nie w `services`.** Dziś `PUT` waliduje dwa razy: `parseBody(BoardBodySchema)` w trasie i `parseBoard` w serwisie. Odpowiedź HTTP na zły body pochodzi z `parseBody`, więc `DomainError` z serwisu był dla HTTP martwy. Po migracji: `router` → `parseBody(BoardBodySchema)` → `normalizeBoard` (trim tytułów kafelków) → `services.saveBoard(board: Board)` (czysty zapis). Odpowiedź 400 na zły body jest identyczna jak dziś.
3. **Walidacja przy odczycie zostaje** (zachowanie 1:1: uszkodzona zawartość pliku daje 400 z komunikatem z `parseBoard`). Repozytorium zwraca surowy `StoredBoard` (`{ tiles: unknown; … }`, dawny `BoardDoc` z `types/records`), serwis przekazuje go dalej, a `schemas.storedBoardToDto` waliduje (`parseBoard`) i zamienia `null` na `EMPTY_BOARD`. Konsekwencja: `services.getBoard` i `saveBoard` to cienkie przekaźniki (reguł domeny tu nie ma, YAGNI), a testy reguł walidacji lądują w `schemas.test.ts`.
4. **Nazwy:** DTO `BoardDto`, `BoardTileDto`, `BoardFrameDto`, `BoardEdgeDto` (jak `UserTagDto`); modele `Board`, `BoardTile`, `BoardFrame`, `BoardEdge`. W UI importy z aliasem (`BoardDto as Board`), jak w `ActivityLog.tsx:17` (`CalendarEventDto as CalendarEvent`): zmiana czysto mechaniczna, bez ruszania ciał komponentów.
5. **`BoardBox`** (prostokąt `x, y, w, h` na płótnie) zostaje w UI: `src/features/board/boardTypes.ts` ma po zmianie tylko ten typ (spec: „`src/features/board/` bez typów domeny”).
6. **Repozytorium to jeden plik** `repositories.ts` (przenosiny 1:1 z `lib/nexus/repos/board.ts`, 35 linii). Jedyna zmiana: typ zwracany `StoredBoard` z `./models` zamiast `BoardDoc` z `@/types/records` (R7 zabrania `contexts/**` → `types/records`).

**Poza zakresem:** format `Board/board.json` (bez zmian: `{ tiles, frames, edges, updatedAt }`), zachowanie UI, `mind-maps` (PR 9), `README.md` i `STAN-W-KODZIE.md` (PR 10).

```ts
// @vitest-environment node
import { describe, expect, it } from "vitest";
import { DomainError } from "@/lib/errors";
import {
  BoardBodySchema,
  EMPTY_BOARD,
  normalizeBoard,
  parseBoard,
  storedBoardToDto,
  toBoardDto,
} from "@/contexts/bc_board/schemas";

const tile = {
  id: "t1",
  title: "  Kafelek  ",
  note: "",
  x: 0,
  y: 0,
  w: 160,
  h: 72,
  frame: null,
};

describe("parseBoard", () => {
  it("trims a tile title and accepts an empty note", () => {
    const board = parseBoard({ tiles: [tile], frames: [], edges: [] });
    expect(board.tiles[0].title).toBe("Kafelek");
  });

  it("rejects an unknown frame, a small tile, and a bad arrow", () => {
    expect(() =>
      parseBoard({ tiles: [{ ...tile, frame: "missing" }], frames: [], edges: [] })
    ).toThrow(DomainError);
    expect(() => parseBoard({ tiles: [{ ...tile, w: 159 }], frames: [], edges: [] })).toThrow(
      /too small|160/
    );
    const emptyIdBoard = () => parseBoard({ tiles: [{ ...tile, id: "" }], frames: [], edges: [] });
    expect(emptyIdBoard).toThrow(DomainError);
    try {
      emptyIdBoard();
    } catch (e) {
      expect((e as DomainError).message).not.toMatch(/too small/);
    }
    expect(() =>
      parseBoard({
        tiles: [tile],
        frames: [],
        edges: [{ id: "e", source: "t1", target: "t1" }],
      })
    ).toThrow(/two different tiles/);
  });

  it("rejects a duplicate id", () => {
    expect(() =>
      parseBoard({
        tiles: [tile],
        frames: [],
        edges: [{ id: "t1", source: "t1", target: "t1" }],
      })
    ).toThrow(/Duplicate id/);
  });

  it("rejects a blank frame title", () => {
    const blankFrameTitle = () =>
      parseBoard({
        tiles: [],
        frames: [{ id: "f", title: "   ", x: 0, y: 0, w: 200, h: 120 }],
        edges: [],
      });
    expect(blankFrameTitle).toThrow(DomainError);
    try {
      blankFrameTitle();
    } catch (e) {
      expect((e as DomainError).message).not.toMatch(/too small/);
    }
  });

  it("reports DomainError with status 400", () => {
    try {
      parseBoard({ tiles: [{ ...tile, w: 10 }], frames: [], edges: [] });
      expect.unreachable();
    } catch (e) {
      expect((e as DomainError).statusCode).toBe(400);
      expect((e as DomainError).message).toBe("Tile or frame is too small");
    }
  });
});

describe("normalizeBoard", () => {
  it("trims tile titles and keeps frames and edges as they are", () => {
    const body = BoardBodySchema.parse({
      tiles: [tile],
      frames: [{ id: "f1", title: "Rama", x: 0, y: 0, w: 200, h: 120 }],
      edges: [],
    });
    const board = normalizeBoard(body);
    expect(board.tiles[0].title).toBe("Kafelek");
    expect(board.tiles[0].frame).toBeNull();
    expect(board.frames).toEqual([{ id: "f1", title: "Rama", x: 0, y: 0, w: 200, h: 120 }]);
    expect(board.edges).toEqual([]);
  });
});

describe("toBoardDto", () => {
  it("copies tiles, frames and edges", () => {
    const board = { tiles: [{ ...tile, title: "Kafelek" }], frames: [], edges: [] };
    expect(toBoardDto(board)).toEqual(board);
  });
});

describe("storedBoardToDto (odczyt z pliku)", () => {
  it("returns an empty board for a missing file", () => {
    expect(storedBoardToDto(null)).toEqual({ tiles: [], frames: [], edges: [] });
    expect(EMPTY_BOARD).toEqual({ tiles: [], frames: [], edges: [] });
  });

  it("validates and normalizes stored data", () => {
    const dto = storedBoardToDto({ tiles: [tile], frames: [], edges: [] });
    expect(dto.tiles[0].title).toBe("Kafelek");
  });

  it("throws DomainError 400 for a corrupt stored board", () => {
    expect(() => storedBoardToDto({ tiles: "x", frames: [], edges: [] })).toThrow(DomainError);
    expect(() =>
      storedBoardToDto({ tiles: undefined, frames: undefined, edges: undefined })
    ).toThrow(DomainError);
  });
});
```

Walidacja z `boardSchema.ts` przechodzi 1:1 (zod, `superRefine`, mapowanie `too_small`).

```ts
import { z } from "zod";
import { DomainError } from "@/lib/errors";
import type { Board, StoredBoard } from "./models";

// ── wejście (zod) ──────────────────────────────────────────────────────────────

export const BoardBodySchema = z
  .object({
    tiles: z.array(
      z.object({
        id: z.string().min(1),
        title: z.string().max(200),
        note: z.string().max(10000),
        x: z.number().finite(),
        y: z.number().finite(),
        w: z.number().finite().min(160),
        h: z.number().finite().min(72),
        frame: z.string().min(1).nullable(),
      })
    ),
    frames: z.array(
      z.object({
        id: z.string().min(1),
        title: z.string().trim().min(1).max(200),
        x: z.number().finite(),
        y: z.number().finite(),
        w: z.number().finite().min(200),
        h: z.number().finite().min(120),
      })
    ),
    edges: z.array(
      z.object({
        id: z.string().min(1),
        source: z.string().min(1),
        target: z.string().min(1),
      })
    ),
  })
  .superRefine((board, ctx) => {
    const ids = [...board.tiles, ...board.frames, ...board.edges].map((item) => item.id);
    if (new Set(ids).size !== ids.length) {
      ctx.addIssue({ code: z.ZodIssueCode.custom, message: "Duplicate id" });
    }
    const frameIds = new Set(board.frames.map((frame) => frame.id));
    const tileIds = new Set(board.tiles.map((tile) => tile.id));
    for (const tile of board.tiles) {
      if (tile.frame && !frameIds.has(tile.frame)) {
        ctx.addIssue({ code: z.ZodIssueCode.custom, message: "Unknown frame" });
      }
    }
    for (const edge of board.edges) {
      if (edge.source === edge.target || !tileIds.has(edge.source) || !tileIds.has(edge.target)) {
        ctx.addIssue({
          code: z.ZodIssueCode.custom,
          message: "Edge must join two different tiles",
        });
      }
    }
  });

export type BoardBody = z.infer<typeof BoardBodySchema>;

/** Normalizacja zwalidowanego body: przycina tytuły kafelków (tytuły ramek przycina sam schemat). */
export function normalizeBoard(body: BoardBody): Board {
  return {
    tiles: body.tiles.map((tile) => ({
      ...tile,
      title: tile.title.trim(),
      frame: tile.frame ?? null,
    })),
    frames: body.frames,
    edges: body.edges,
  };
}

/** Waliduje dowolne dane (np. zawartość pliku); zły kształt to `DomainError(400)`. */
export function parseBoard(input: unknown): Board {
  const result = BoardBodySchema.safeParse(input);
  if (!result.success) {
    const tooSmall = result.error.issues.find(
      (issue) => issue.code === "too_small" && issue.origin === "number"
    );
    const message = tooSmall
      ? "Tile or frame is too small"
      : (result.error.issues[0]?.message ?? "Invalid board");
    throw new DomainError(message, 400);
  }
  return normalizeBoard(result.data);
}

// ── wyjście (DTO) ──────────────────────────────────────────────────────────────

export type BoardTileDto = {
  id: string;
  title: string;
  note: string;
  x: number;
  y: number;
  w: number;
  h: number;
  frame: string | null;
};

export type BoardFrameDto = {
  id: string;
  title: string;
  x: number;
  y: number;
  w: number;
  h: number;
};

export type BoardEdgeDto = {
  id: string;
  source: string;
  target: string;
};

export type BoardDto = {
  tiles: BoardTileDto[];
  frames: BoardFrameDto[];
  edges: BoardEdgeDto[];
};

export const EMPTY_BOARD: BoardDto = { tiles: [], frames: [], edges: [] };

export function toBoardDto(board: Board): BoardDto {
  return {
    tiles: board.tiles.map((tile) => ({ ...tile })),
    frames: board.frames.map((frame) => ({ ...frame })),
    edges: board.edges.map((edge) => ({ ...edge })),
  };
}

/** Odczyt z pliku: brak pliku → pusta tablica; uszkodzona zawartość → `DomainError(400)`. */
export function storedBoardToDto(stored: StoredBoard | null): BoardDto {
  if (stored === null) return EMPTY_BOARD;
  return toBoardDto(parseBoard(stored));
}
```

```ts
export async function putBoard(request: NextRequest): Promise<NextResponse> {
  const parsed = await parseBody(request, BoardBodySchema);
  if (parsed.success === false) return parsed.response;
  try {
    const saved = await services.saveBoard(normalizeBoard(parsed.data));
    return NextResponse.json(toBoardDto(saved));
  } catch (err) {

}
```
